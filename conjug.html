let running = true;

function getQuestionText() {
  const selectors = [
    '.question-text',
    '[data-question]',
    'h1',
    'h2',
    '.prompt',
    '[role="heading"]'
  ];
  for (let s of selectors) {
    const el = document.querySelector(s);
    if (el && el.innerText.trim()) return el.innerText.trim();
  }
  return null;
}

function autoTranslate() {
  if (!running) return;

  const input = document.getElementById('answer-input');
  if (!input) return;

  const questionText = getQuestionText();
  if (!questionText) return;

  let translation = translations[questionText];
  if (!translation) {
    const lower = questionText.toLowerCase();
    for (let key in translations) {
      if (key.toLowerCase() === lower) {
        translation = translations[key];
        break;
      }
    }
  }

  if (!translation) return;

  const first = translation.split('/')[0];
  input.value = first;
  input.dispatchEvent(new Event('input', { bubbles: true }));
  input.dispatchEvent(new Event('change', { bubbles: true }));
}

const observer = new MutationObserver(autoTranslate);

function start() {
  if (running) return;
  running = true;
  observer.observe(document.body, { childList: true, subtree: true });
  updateNotice();
  autoTranslate();
}

function stop() {
  if (!running) return;
  running = false;
  observer.disconnect();
  updateNotice();
}

function toggle() {
  if (running) stop();
  else start();
}

function updateNotice() {
  notice.style.background = running ? "green" : "red";
  notice.textContent = running ? "Running (Press ESC to Stop)" : "Stopped (Press ESC to Start)";
}

const notice = document.createElement('div');
notice.style.position = 'fixed';
